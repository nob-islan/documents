# 単体テスト作成

単体テストの書き方およびカバレッジの確認方法を説明します。

cf. https://spring.pleiades.io/spring-boot/reference/testing/spring-boot-applications.html

## 事前準備

下記設定を追記します:

- application-test.properties

```properties
# エンティティクラスからスキーマを自動生成しない
spring.jpa.hibernate.ddl-auto=none
# h2db接続設定
spring.datasource.url=jdbc:h2:mem:testdb;DB_CLOSE_DELAY=-1
spring.datasource.driver-class-name=org.h2.Driver
spring.datasource.username=sa
spring.datasource.password=
spring.jpa.database-platform=org.hibernate.dialect.H2Dialect
```

- pom.xml

```xml
		<!-- https://mvnrepository.com/artifact/com.h2database/h2 -->
        <dependency>
            <groupId>com.h2database</groupId>
            <artifactId>h2</artifactId>
            <scope>test</scope>
        </dependency>
```

- XxxApplicationTests.java

```java
// test配下の自動生成済みテストクラスに下記を追記
@ActiveProfiles("test")
```

## 作成手順

下記テーブルに対してデータを取得 / 登録する API を想定しています:

### controller テスト作成

```java
package nob.example.easyapp.controller;

import java.util.List;

import org.junit.jupiter.api.Test;
import org.mockito.Mockito;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.servlet.WebMvcTest;
import org.springframework.http.MediaType;
import org.springframework.test.context.bean.override.mockito.MockitoBean;
import org.springframework.test.web.servlet.MockMvc;
import org.springframework.test.web.servlet.MvcResult;

import com.fasterxml.jackson.databind.ObjectMapper;

import nob.example.easyapp.controller.model.UserRegistRequest;
import nob.example.easyapp.controller.model.UserRegistResponse;
import nob.example.easyapp.controller.model.UserSearchRequest;
import nob.example.easyapp.controller.model.UserSearchResponse;
import nob.example.easyapp.dto.UsersDto;
import nob.example.easyapp.service.UserinfoService;
import nob.example.easyapp.service.model.UserRegistInModel;
import nob.example.easyapp.service.model.UserRegistOutModel;
import nob.example.easyapp.service.model.UserSearchInModel;
import nob.example.easyapp.service.model.UserSearchOutModel;

import static org.assertj.core.api.Assertions.assertThat;
import static org.junit.jupiter.api.Assertions.fail;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.get;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.post;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.status;

@WebMvcTest
public class UserinfoControllerImplTest {

    @Autowired
    private MockMvc mockMvc;

    @Autowired
    private ObjectMapper objectMapper;

    @MockitoBean
    private UserinfoService userinfoService;

    /**
     * searchのテスト 正常系
     */
    @Test
    void test_search_success() {

        // 入力値の作成
        UserSearchRequest request = new UserSearchRequest("nob");

        // serviceのモック化
        Mockito.when(userinfoService.search(new UserSearchInModel(request.getUsername()))).thenReturn(
                new UserSearchOutModel(List.of(new UsersDto(1, "testnob1", 13), new UsersDto(2, "testnob2", 706))));

        try {
            // API呼び出し
            MvcResult result = mockMvc.perform(get("/api/v1/user")
                    .queryParam("username", request.getUsername())
                    .contentType(MediaType.APPLICATION_JSON))
                    .andExpect(status().isOk())
                    .andReturn();
            // 結果のassert
            assertThat(result.getResponse().getContentAsString())
                    .isEqualTo(objectMapper.writeValueAsString(new UserSearchResponse(
                            List.of(new UsersDto(1, "testnob1", 13), new UsersDto(2, "testnob2", 706)))));
        } catch (Exception e) {
            e.printStackTrace();
            fail(); // 例外をキャッチしたらテスト失敗
        }
    }

    /**
     * registのテスト 正常系
     */
    @Test
    void test_regist_success() {

        // 入力値の作成
        UserRegistRequest request = new UserRegistRequest(new UsersDto(null, "nob", 13));

        // serviceのモック化
        Mockito.when(userinfoService.regist(new UserRegistInModel(
                new UsersDto(null, request.getUsersDto().getUsername(), request.getUsersDto().getAge()))))
                .thenReturn(new UserRegistOutModel(true));

        try {
            // API呼び出し
            MvcResult result = mockMvc.perform(post("/api/v1/user")
                    .content(objectMapper.writeValueAsString(request))
                    .contentType(MediaType.APPLICATION_JSON))
                    .andExpect(status().isOk())
                    .andReturn();
            // 結果のassert
            assertThat(result.getResponse().getContentAsString())
                    .isEqualTo(objectMapper.writeValueAsString(new UserRegistResponse(true)));
        } catch (Exception e) {
            e.printStackTrace();
            fail(); // 例外をキャッチしたらテスト失敗
        }
    }
}
```

### service テスト作成

```java
package nob.example.easyapp.service;

import static org.assertj.core.api.Assertions.assertThat;
import static org.junit.jupiter.api.Assertions.fail;

import java.util.List;

import org.junit.jupiter.api.Test;
import org.mockito.Mockito;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.test.context.ActiveProfiles;
import org.springframework.test.context.bean.override.mockito.MockitoBean;

import nob.example.easyapp.dto.UsersDto;
import nob.example.easyapp.repository.UsersRepository;
import nob.example.easyapp.repository.entity.Users;
import nob.example.easyapp.service.model.UserRegistInModel;
import nob.example.easyapp.service.model.UserRegistOutModel;
import nob.example.easyapp.service.model.UserSearchInModel;
import nob.example.easyapp.service.model.UserSearchOutModel;

@SpringBootTest
@ActiveProfiles("test")
public class UserinfoServiceImplTest {

    @Autowired
    private UserinfoService userinfoService;

    @MockitoBean
    private UsersRepository usersRepository;

    /**
     * searchのテスト 正常系
     */
    @Test
    void test_search_success() {

        // 入力値の作成
        UserSearchInModel inModel = new UserSearchInModel("nob");

        // repositoryのモック化
        Users users01 = new Users(1, "testnob1", 13);
        Users users02 = new Users(2, "testnob2", 706);
        Mockito.when(usersRepository.selectByUsername(inModel.getUsername()))
                .thenReturn(List.of(users01, users02));

        try {
            // service呼び出し
            UserSearchOutModel outModel = userinfoService.search(inModel);
            // 結果のassert
            assertThat(outModel.getUsersDtoList().size()).isEqualTo(2);
            assertThat(outModel.getUsersDtoList().get(0))
                    .isEqualTo(new UsersDto(users01.getId(), users01.getUsername(), users01.getAge()));
            assertThat(outModel.getUsersDtoList().get(1))
                    .isEqualTo(new UsersDto(users02.getId(), users02.getUsername(), users02.getAge()));
        } catch (Exception e) {
            e.printStackTrace();
            fail(); // 例外をキャッチしたらテスト失敗
        }
    }

    /**
     * registのテスト 正常系
     */
    @Test
    void test_regist_success() {

        // 入力値の作成
        UserRegistInModel inModel = new UserRegistInModel(new UsersDto(null, "nob", 13));

        // repositoryのモック化
        Mockito.when(usersRepository
                .save(new Users(null, inModel.getUsersDto().getUsername(), inModel.getUsersDto().getAge())))
                .thenReturn(new Users());

        try {
            // service呼び出し
            UserRegistOutModel outModel = userinfoService.regist(inModel);
            // 結果のassert
            assertThat(outModel.isSuccess()).isTrue();
        } catch (Exception e) {
            e.printStackTrace();
            fail(); // 例外をキャッチしたらテスト失敗
        }
    }
}
```

### repository テスト作成

#### テストデータ準備

`test/resources/repository/users` 配下にテストデータ投入向けの SQL を用意します:

- schema.sql

```sql
CREATE TABLE IF NOT EXISTS users(
    id INT PRIMARY KEY AUTO_INCREMENT
    , username VARCHAR(20) NOT NULL
    , age INT NOT NULL
);
```

- data.sql

```sql
INSERT INTO users (
    username
    , age
) VALUES (
    'nob'
    , 13
);
```

#### テストケース作成

```java
package nob.example.easyapp.repository;

import static org.assertj.core.api.Assertions.assertThat;
import static org.junit.jupiter.api.Assertions.fail;

import java.util.List;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.orm.jpa.DataJpaTest;
import org.springframework.test.context.ActiveProfiles;
import org.springframework.test.context.TestPropertySource;

import nob.example.easyapp.repository.entity.Users;

/**
 * UsersRepositoryのテストクラスです。
 *
 * @author nob
 */
@DataJpaTest
@ActiveProfiles("test")
@TestPropertySource(properties = {
        "spring.sql.init.schema-locations=classpath:/repository/users/schema.sql",
        "spring.sql.init.data-locations=classpath:/repository/users/data.sql"
})
public class UsersRepositoryTest {

    @Autowired
    private UsersRepository usersRepository;

    /**
     * selectByUsernameのテスト
     */
    @Test
    public void test_selectByUsername() {

        try {
            // repository呼び出し
            List<Users> usersList = usersRepository.selectByUsername("nob");
            // 結果のassert
            assertThat(usersList.size()).isEqualTo(1);
            assertThat(usersList.get(0).getId()).isEqualTo(1);
            assertThat(usersList.get(0).getUsername()).isEqualTo("nob");
            assertThat(usersList.get(0).getAge()).isEqualTo(13);
        } catch (Exception e) {
            e.printStackTrace();
            fail(); // 例外をキャッチしたらテスト失敗
        }
    }
}
```

## カバレッジ出力

cf. https://www.jacoco.org/jacoco/trunk/doc/maven.html

### 設定

下記設定を追記します:

- pom.xml

```xml
			<plugin>
				<groupId>org.jacoco</groupId>
				<artifactId>jacoco-maven-plugin</artifactId>
				<version>0.8.14</version>
				<executions>
					<execution>
						<goals>
							<goal>prepare-agent</goal>
						</goals>
					</execution>
					<execution>
						<id>report</id>
						<phase>test</phase>
						<goals>
							<goal>report</goal>
						</goals>
					</execution>
				</executions>
			</plugin>
```

```xml
	<reporting>
		<plugins>
			<plugin>
				<groupId>org.jacoco</groupId>
				<artifactId>jacoco-maven-plugin</artifactId>
				<reportSets>
					<reportSet>
						<reports>
						<!-- select non-aggregate reports -->
						<report>report</report>
						</reports>
					</reportSet>
				</reportSets>
			</plugin>
		</plugins>
	</reporting>
```
